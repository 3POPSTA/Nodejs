const http = require("http");

const fs = require("fs");

const path = require("path");



const server = http.createServer((request,response)=>{

    const filePath = path.join(__dirname,request.url);



    fs.access(filePath,fs.constants.F_OK,(error)=>{

        if(error){

            response.statusCode = 404;

            response.end("File not found");

            return;

        }

    })



    fs.stat(filePath,(error,stats)=>{

        if(error){

            response.statusCode = 500;

            response.end("Server error");

            return;

        }



        response.setHeader('Content-Length', stats.size);

        response.setHeader('Content-Type', 'application/octet-stream');



        const stream = fs.createReadStream(filePath);



        stream.on("error", (error)=> {

            console.log("Error reading file: ",error);

            if(!response.headersSent){

                response.statusCode = 500;

                response.end('Error reading file');

            }

        });

        stream.pipe(response);

    });

});

const PORT = 3000;

server.listen(PORT, () => {

    console.log(`File server running at http://localhost:${PORT}/`);

});

This code has critical security vulnerabilities and logic bugs that will cause crashes.

1. Severe Security Risk: Path Traversal
path.join(__dirname, request.url) allows anyone to look at files outside your project folder. If a user requests a URL like http://localhost:3000/../../etc/passwd, your server will read and expose sensitive system files. You must verify that the requested path stays inside your project directory.

2. Race Condition and Multiple Responses
fs.access and fs.stat run at the same time independently. fs.access does not block fs.stat from running. If a file is missing, fs.access sends a 404 response, but fs.stat immediately runs right after it, fails, and tries to send a 500 response. This crashes your app because Node.js does not allow you to send headers or responses twice for a single request. You only need fs.stat (or fs.lstat) to check for existence and file details at the same time.

3. Missing Directory Check
If a user requests a folder path instead of a file, fs.stat treats the folder as valid. Trying to stream a directory causes a server error. You must check that the path is actually a file using stats.isFile().

Corrected Code
This fixed version prevents path traversal, removes the redundant check, handles missing files and directories cleanly, and stops server crashes:

JavaScript
const http = require("http");
const fs = require("fs");
const path = require("path");

const server = http.createServer((request, response) => {
    // Prevent path traversal by resolving the absolute path and checking boundaries
    const filePath = path.resolve(__dirname, "." + request.url);

    if (!filePath.startsWith(__dirname)) {
        response.statusCode = 403;
        response.end("Access denied");
        return;
    }

    fs.stat(filePath, (error, stats) => {
        // If the file does not exist, or if it is a folder instead of a file, return 404
        if (error || !stats.isFile()) {
            response.statusCode = 404;
            response.end("File not found");
            return;
        }

        response.setHeader('Content-Length', stats.size);
        response.setHeader('Content-Type', 'application/octet-stream');

        const stream = fs.createReadStream(filePath);

        stream.on("error", (error) => {
            console.log("Error reading file: ", error);
            if (!response.headersSent) {
                response.statusCode = 500;
                response.end('Error reading file');
            }
        });

        stream.pipe(response);
    });
});

const PORT = 3000;
server.listen(PORT, () => {
    console.log(`File server running at http://localhost:${PORT}/`);
});
